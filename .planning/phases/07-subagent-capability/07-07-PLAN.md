    Implement YAML frontmatter parsing via Node.js inline script (consistent with rest of codebase):
    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    AGENTS_DIR="$(cd "$SCRIPT_DIR/.." && pwd)/agents"

    if [[ ! -d "$AGENTS_DIR" ]]; then
      echo "agents/ directory not found at $AGENTS_DIR" >&2
      exit 1
    fi

    node --input-type=module <<EOF
    import fs from 'fs';
    import path from 'path';
    import yaml from 'js-yaml';

    const agentsDir = '$AGENTS_DIR';
    const entries = fs.readdirSync(agentsDir, { withFileTypes: true })
      .filter(e => e.isDirectory());

    const profiles = [];
    for (const entry of entries) {
      const mdPath = path.join(agentsDir, entry.name, 'AGENT.md');
      if (!fs.existsSync(mdPath)) continue;
      const content = fs.readFileSync(mdPath, 'utf8');
      const match = content.match(/^---\n([\s\S]*?)\n---/);
      if (!match) continue;
      try {
        const fm = yaml.load(match[1]);
        profiles.push(fm);
      } catch { continue; }
    }

    const col = (s, n) => String(s ?? '-').padEnd(n).slice(0, n);
    console.log(col('NAME', 20) + col('MODEL', 30) + col('MAX_TOKENS', 12) + col('CAN_SPAWN', 11) + 'DESCRIPTION');
    console.log('-'.repeat(100));
    for (const p of profiles) {
      const tools = Array.isArray(p.tools) ? p.tools.join(',') : '-';
      const desc = p.description ?? '-';
      console.log(col(p.name, 20) + col(p.model, 30) + col(p.max_tokens, 12) + col(p.can_spawn, 11) + desc.slice(0, 60));
    }
    EOF
    ```

    Note: `js-yaml` must be available in the project's node_modules. Since scripts/profiles.sh runs from the repo root, use `node --input-type=module` with the import.
