    ```typescript
    import { marked } from 'marked';
    import { markedTerminal } from 'marked-terminal';
    marked.use(markedTerminal());
    export function renderMarkdown(text: string): string {
      return marked(text) as string;
    }
    ```

    **tui/src/hooks/useSSE.ts:**
    State: connectionStatus ('connected' | 'reconnecting' | 'disconnected'), last SSE event.
    - On mount: call streamSSE(url, token) in an async loop, dispatch each event via onEvent callback
    - On error or stream end: set status 'reconnecting', wait with exponential backoff (1s, 2s, 4s, max 30s), retry
    - Expose reconnect() function (resets backoff, triggers immediate retry)
    - Return: { connectionStatus, reconnect }
    - Use useEffect with cleanup that sets a cancelled flag to stop the retry loop on unmount

    **tui/src/hooks/useChat.ts:**
    State: messages array, in-progress streaming message.
    Message shape:
    ```typescript
    interface ChatMessage {
      id: string;  // crypto.randomUUID()
      role: 'user' | 'agent';
      text: string;           // accumulated text
      thinking?: string;      // accumulated thinking text
      tools: ToolCall[];      // tool calls in order
      timestamp: Date;
      isStreaming: boolean;
      turns?: number;
      tokens?: number;
    }
    interface ToolCall { name: string; input: Record<string,unknown>; result?: string; }
    ```
    - onAgentEvent(event: AgentEvent): dispatch function:
      - 'token': append to current streaming message text
      - 'thinking': append to current streaming message thinking
      - 'tool_start': push new ToolCall to current message tools
      - 'tool_result': update last tool in current message with result
      - 'done': finalize current streaming message (isStreaming=false, set turns/tokens from result)
      - 'error': append error message
      - 'idle': finalize any in-progress message
    - addUserMessage(text: string): append user message to messages
    - clearMessages(): reset messages to []
    - Return: { messages, onAgentEvent, addUserMessage, clearMessages }

    **tui/src/hooks/useHealth.ts:**
    - Polls `./talos health --json` by spawning child process via Node `exec` (import from node:child_process)
    - Parses JSON output into HealthStatus { db, mcp, llm, embed }
    - Polls every 30 seconds, also on mount
    - Return: { health: HealthStatus | null }
    - Use useEffect with setInterval cleanup
